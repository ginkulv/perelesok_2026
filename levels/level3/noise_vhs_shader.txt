shader_type canvas_item;

// Объявляем SCREEN_TEXTURE для Godot 4
uniform sampler2D SCREEN_TEXTURE : hint_screen_texture, filter_linear_mipmap;

uniform float vhs_strength : hint_range(0.0, 0.5) = 0.1;
uniform float jitter_amount : hint_range(0.0, 0.05) = 0.01;
uniform float color_bleed : hint_range(0.0, 0.03) = 0.008;
uniform float scanline_intensity : hint_range(0.0, 0.3) = 0.1;
uniform float noise_strength : hint_range(0.0, 0.5) = 0.08;
uniform float noise_speed : hint_range(0.0, 3.0) = 0.5;

// Параметры для чёрно-белого эффекта
uniform float desaturate_start : hint_range(0.0, 0.3) = 0.08;
uniform float desaturate_end : hint_range(0.0, 0.5) = 0.2;

// Исправленная функция random
float random(vec2 st) {
    return fract(sin(dot(st.xy, vec2(12.9898,78.233))) * 43758.5453123);
}

// Функция для преобразования в чёрно-белый
float grayscale(vec3 color) {
    return dot(color, vec3(0.299, 0.587, 0.114));
}

void fragment() {
    // Используем SCREEN_UV для получения того, что находится за спрайтом
    vec2 uv = SCREEN_UV;
    float t = TIME;
    
    // Сохраняем оригинальный цвет до искажений
    vec4 original_color = texture(SCREEN_TEXTURE, uv);
    
    // VHS эффекты применяются к экранным координатам
    float jitter = random(vec2(floor(uv.y * 200.0), floor(t * 2.0))) * jitter_amount;
    uv.y += jitter;
    
    float shake = sin(t * 30.1) * 0.0001;
    uv.x += shake;
    
    // Color bleeding на экранной текстуре
    vec4 color_r = texture(SCREEN_TEXTURE, uv + vec2(color_bleed, 0.0));
    vec4 color_g = texture(SCREEN_TEXTURE, uv);
    vec4 color_b = texture(SCREEN_TEXTURE, uv - vec2(color_bleed, 0.0));
    vec4 color = vec4(color_r.r, color_g.g, color_b.b, 1.0);
    
    // Scanlines
    float scanline = sin(uv.y * 800.0 + t * 20.0) * scanline_intensity;
    color.rgb -= scanline;
    
    // Noise - исправлено использование random
    float noise = random(uv + floor(t * noise_speed)) * noise_strength;
    color.rgb += vec3(noise);
    
    // ПЛАВНЫЙ ПЕРЕХОД В ЧЁРНО-БЕЛЫЙ
    float desaturate_factor;
    
    if (noise_strength <= desaturate_start) {
        desaturate_factor = 0.0;
    } 
    else if (noise_strength >= desaturate_end) {
        desaturate_factor = 1.0;
    } 
    else {
        desaturate_factor = (noise_strength - desaturate_start) / (desaturate_end - desaturate_start);
        desaturate_factor = pow(desaturate_factor, 1.2);
    }
    
    // Получаем чёрно-белую версию
    float gray = grayscale(color.rgb);
    vec3 desaturated_color = vec3(gray);
    
    // Смешиваем цветной и чёрно-белый варианты
    vec3 final_color = mix(color.rgb, desaturated_color, desaturate_factor);
    
    // Небольшой эффект мерцания при сильном шуме
    if (noise_strength > desaturate_start) {
        float flicker = 1.0 + (random(uv + floor(t * 10.0)) - 0.5) * noise_strength * 0.2;
        final_color *= flicker;
    }
    
    COLOR = vec4(final_color, 1.0);
}
